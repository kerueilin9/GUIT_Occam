\begin{algorithm}[t]
\caption{Generate ISP-Assisted Form Test Cases}
\label{alg:isp-generation}

\SetKwInOut{KwIn}{Input}
\SetKwInOut{KwOut}{Output}

\KwIn{
$\mathit{discoveries}$: recorded type actions, each containing $\{element\_id,\ original\_value,\ obs\_text\}$;\\
$\mathit{gherkin}$: original Gherkin task;\\
$\mathit{field\_hints}$: optional field hints;\\
$\mathit{max\_fields}$: maximum number of aligned fields;\\
$\mathit{max\_cases}$: maximum number of generated form test cases
}
\KwOut{
$\mathit{cases}$: structured form test cases, each containing $\{name,\ expected\_hint,\ inputs\}$
}

\BlankLine

$\mathit{fillSteps} \leftarrow \Call{ExtractFillSteps}{\mathit{gherkin.when}}$\;

\If{$\mathit{discoveries} = \emptyset$}{
    \Return $\emptyset$\;
}

$\mathit{logicalFields} \leftarrow \emptyset$\;
$\mathit{matchCounts} \leftarrow \emptyset$\;

\ForEach{$d \in \mathit{discoveries}$}{
    $m \leftarrow \Call{ExtractFieldMetadata}{d.element\_id,\ d.obs\_text,\ \mathit{field\_hints}}$\;
    $m.\mathit{original\_value} \leftarrow d.\mathit{original\_value}$\;

    $(f,\ idx) \leftarrow \Call{MatchToFillStep}{m,\ \mathit{fillSteps},\ \mathit{matchCounts}}$\;

    \If{$f = \texttt{null} \text{ and } \Call{OrderFallbackApplicable}{\mathit{discoveries},\ \mathit{fillSteps}}$}{
        $(f,\ idx) \leftarrow \Call{OrderFallback}{d,\ \mathit{fillSteps}}$\;
    }

    \If{$f \neq \texttt{null}$}{
        $m.\mathit{label} \leftarrow f.\mathit{display}$\;
        $m.\mathit{input\_type} \leftarrow \Call{RefineInputType}{m,\ f}$\;
        $\Call{UpdateLogicalField}{\mathit{logicalFields},\ m,\ f,\ idx}$\;
        $\mathit{matchCounts}[m.\mathit{label}] \leftarrow \mathit{matchCounts}[m.\mathit{label}] + 1$\;
    }
}

$\mathit{logicalFields} \leftarrow \Call{FinalizeLogicalFields}{\mathit{logicalFields},\ \mathit{max\_fields}}$\;
$\mathit{fieldMetas} \leftarrow \Call{PrepareFieldMetadata}{\mathit{logicalFields}}$\;

$\mathit{rawCases} \leftarrow \Call{GenerateWithLLM}{\mathit{fieldMetas},\ \mathit{gherkin},\ \mathit{max\_cases}}$\;
$\mathit{cases} \leftarrow \Call{ParseAndNormalizeCases}{\mathit{rawCases},\ \mathit{fieldMetas}}$\;

\If{$\mathit{cases} = \emptyset$}{
    $\mathit{cases} \leftarrow \Call{GenerateFallbackCases}{\mathit{fieldMetas},\ \mathit{gherkin},\ \mathit{max\_cases}}$\;
}

\ForEach{$c \in \mathit{cases}$}{
    $c.\mathit{expected\_hint} \leftarrow \Call{NormalizeOrInferExpectedHint}{c,\ \mathit{fieldMetas}}$\;
}

\Return $\mathit{cases}$\;
\end{algorithm}
